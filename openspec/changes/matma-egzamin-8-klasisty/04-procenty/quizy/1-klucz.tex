\textbf{Zadanie 1.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $160:10=\mathbf{16}$.
  \item $25\%=\frac{1}{4}$, $60:4=\mathbf{15}$.
\end{enumerate}

\textbf{Zadanie 2.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $10\%=4$, $5\%=2$, razem $\mathbf{6}$.
  \item $10\%=6$, $3\cdot 6=18$, $5\%=3$, razem $18+3=\mathbf{21}$.
\end{enumerate}

\textbf{Zadanie 3.} Błąd: $5\%$ to $\frac{1}{20}$, a nie $\frac{1}{5}$ ($\frac{1}{5}$ to $20\%$). Dobry wynik: $120:20=\mathbf{6\ zł}$ (albo $10\%=12$, połowa to $6$). Punktacja: $1$ pkt za wskazanie błędu, $1$ pkt za wynik.

Typowa pomyłka: mylenie $5\%$ z $\frac{1}{5}$ oraz „$10\%$ to minus $10$”. Jeśli Nadia liczy $15\%$ w słupku, przypomnij klocki $10\%+5\%$.
