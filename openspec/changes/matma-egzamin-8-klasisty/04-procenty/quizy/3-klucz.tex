\textbf{Zadanie 1.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $25\%=\frac{1}{4}$, $80:4=\mathbf{20\ zł}$.
  \item $80-20=\mathbf{60\ zł}$ (albo $80\cdot 0{,}75$).
\end{enumerate}

\textbf{Zadanie 2.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $50\cdot 1{,}2=\mathbf{60\ zł}$ (albo $50+10$).
  \item $90\ \text{zł}$ to $90\%$ ceny, więc $10\%=10$ i cena przed to $\mathbf{100\ zł}$.
\end{enumerate}

\textbf{Zadanie 3.} $1000\cdot 1{,}1=1100\ \text{zł}$, potem $1100\cdot 0{,}9=990\ \text{zł}$. Odpowiedź: $\mathbf{990\ zł}$. Punktacja: $1$ pkt za $1100\ \text{zł}$ (albo mnożnik $1{,}1$), $1$ pkt za $990\ \text{zł}$.

Typowa pomyłka: odpowiedź $1000\ \text{zł}$, czyli „$+10\%$ i $-10\%$ się znoszą”, oraz $90+10\%=99$ w zadaniu 2b. Drugi procent liczymy z nowej ceny.
