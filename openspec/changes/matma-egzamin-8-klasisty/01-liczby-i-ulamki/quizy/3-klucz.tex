\textbf{Zadanie 2.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item Mianownik $70$: $\frac{50}{70}$ vs $\frac{49}{70}$, więc $\frac{5}{7}>\frac{7}{10}$ (\textbf{$>$}).
  \item Ta sama góra, mniejszy mianownik jest większy: $\frac{3}{7}>\frac{3}{8}$ (\textbf{$>$}).
\end{enumerate}

\textbf{Zadanie 3.}

  \item $0{,}60>0{,}57$ (\textbf{$>$}).
\end{enumerate}
