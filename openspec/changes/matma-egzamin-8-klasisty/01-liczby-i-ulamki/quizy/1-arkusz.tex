\vspace{0.4em}

\textbf{Zadanie 1} (2 pkt). Uzupełnij (przeciwna zmienia znak, odwrotność robi fikołka). Pokaż obliczenia.
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item Dla $\dfrac{3}{7}$: liczba przeciwna to \rule{2.5cm}{0.4pt}, odwrotność to \rule{2.5cm}{0.4pt}.
  \item Dla $-4$: liczba przeciwna to \rule{2.5cm}{0.4pt}, odwrotność to \rule{2.5cm}{0.4pt}.
\end{enumerate}
\miejsce{3.8cm}

\textbf{Zadanie 2} (2 pkt). Najpierw zapisz liczbę jako ułamek, potem znajdź odwrotność. Pokaż obliczenia.
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $1\dfrac{2}{5}=\dfrac{\square}{5}$, odwrotność: \rule{2.5cm}{0.4pt}
  \item $0{,}3=\dfrac{\square}{10}$, odwrotność: \rule{2.5cm}{0.4pt}
\end{enumerate}
\miejsce{4.4cm}

\textbf{Zadanie 3} (2 pkt). Znajdź odwrotność. Pokaż obliczenia.
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $0{,}25$: \rule{2.5cm}{0.4pt} \hspace{1cm} b) $-2\dfrac{1}{3}$: \rule{2.5cm}{0.4pt}
\end{enumerate}
\miejsce{4.4cm}
