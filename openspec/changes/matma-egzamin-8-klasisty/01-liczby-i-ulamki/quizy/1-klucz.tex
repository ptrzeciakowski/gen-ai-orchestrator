\naglowekKarty{1}{Klucz: odwrotność}{dla Pawła – nie drukować dla Nadii}

\vspace{0.6em}
\textbf{Zadanie 1.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item Przeciwna: $-\dfrac{3}{7}$, odwrotność: $\dfrac{7}{3}=2\dfrac{1}{3}$.
  \item Przeciwna: $4$, odwrotność: $-\dfrac{1}{4}$ (znak zostaje!).
\end{enumerate}

\textbf{Zadanie 2.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $1\dfrac{2}{5}=\dfrac{7}{5}$, odwrotność: $\dfrac{5}{7}$.
  \item $0{,}3=\dfrac{3}{10}$, odwrotność: $\dfrac{10}{3}=3\dfrac{1}{3}$.
\end{enumerate}

\textbf{Zadanie 3.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $0{,}25=\dfrac{1}{4}$, odwrotność: $4$.
  \item $-2\dfrac{1}{3}=-\dfrac{7}{3}$, odwrotność: $-\dfrac{3}{7}$.
\end{enumerate}
