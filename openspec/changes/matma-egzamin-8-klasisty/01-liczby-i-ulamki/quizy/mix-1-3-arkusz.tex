\textbf{Zadanie 1} (2 pkt). Wyznacz liczbę przeciwną i odwrotność. \textbf{Pokaż obliczenia.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=6pt]
  \item $\dfrac{5}{8}$: przeciwna \rule{2.5cm}{0.4pt}, odwrotność \rule{2.5cm}{0.4pt}
  \item $-6$: przeciwna \rule{2.5cm}{0.4pt}, odwrotność \rule{2.5cm}{0.4pt}
\end{enumerate}
\miejsce{3.8cm}

\textbf{Zadanie 2} (2 pkt). Wpisz $>$, $<$ albo $=$ i uzasadnij. \textbf{Pokaż obliczenia.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=6pt]
  \item $0{,}(7)\ \bigcirc\ 0{,}78$
  \item $\dfrac{7}{12}\ \bigcirc\ \dfrac{5}{8}$ \hspace{0.8cm} (sprowadź do wspólnego mianownika)
\end{enumerate}
\miejsce{4.4cm}

\textbf{Zadanie 3} (2 pkt). Dane są liczby: $a$ = odwrotność liczby $0{,}4$ oraz $b$ = liczba przeciwna do $-2\dfrac{3}{5}$. \textbf{Pokaż obliczenia.}
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=6pt]
  \item Zapisz $a$ i $b$ jako ułamki zwykłe: $a=$ \rule{2.5cm}{0.4pt}, $b=$ \rule{2.5cm}{0.4pt}
  \item Wpisz $>$, $<$ albo $=$: $a\ \bigcirc\ b$
\end{enumerate}
\miejsce{4.4cm}
