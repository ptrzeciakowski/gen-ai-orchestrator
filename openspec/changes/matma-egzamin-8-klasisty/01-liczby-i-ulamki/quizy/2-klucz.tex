\textbf{Zadanie 1.} (pełny zapis: rozkład mianownika i wniosek)
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=4pt]
  \item $\frac{3}{20}$: $20=2^2\cdot 5$, same $2$ i $5$, więc \textbf{skończone}.
  \item $\frac{4}{9}$: $9=3^2$, jest trójka, więc \textbf{nieskończone}.
  \item $\frac{5}{8}$: $8=2^3$, więc \textbf{skończone}.
  \item $\frac{2}{7}$: $7$ jest liczbą pierwszą inną niż $2$ i $5$, więc \textbf{nieskończone}.
\end{enumerate}
