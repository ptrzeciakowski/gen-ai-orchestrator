\textbf{Zadanie 1} (2 pkt). Uzupełnij (przeciwna zmienia znak, odwrotność robi fikołka):
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=6pt]
  \item Dla $\dfrac{3}{7}$: liczba przeciwna to \rule{3cm}{0.4pt}, odwrotność to \rule{3cm}{0.4pt}.
  \item Dla $-4$: liczba przeciwna to \rule{3cm}{0.4pt}, odwrotność to \rule{3cm}{0.4pt}.
\end{enumerate}

\textbf{Zadanie 2} (2 pkt). Najpierw zapisz liczbę jako ułamek, potem znajdź odwrotność:
\begin{enumerate}[label=\alph*), leftmargin=*, itemsep=6pt]
  \item $1\dfrac{2}{5}=\dfrac{\rule{0.8cm}{0.4pt}}{5}$, odwrotność: \rule{3cm}{0.4pt}
  \item $0{,}3=\dfrac{\rule{0.8cm}{0.4pt}}{10}$, odwrotność: \rule{3cm}{0.4pt}
\end{enumerate}

\textbf{Zadanie 3} (2 pkt). Znajdź odwrotność:

\end{enumerate}
